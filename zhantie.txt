// ============================================================
// 🎯 A股内参 - 深度研报生成器 (V12.0 满血扩容版)
// ============================================================

// 1. 🔍 智能数据解包
let root = $input.item.json;
if (root.data && (root.data.hot_sectors || root.data.market_sentiment)) {
    root = root.data;
}

// 2. 提取核心数据源
const sentiment = root.market_sentiment || {};
const portfolio = root.my_portfolio || { stocks: [] };
const sectors = root.hot_sectors || [];
const news = root.news_brief || { items: [] };

// 3. 🛠️ 构建【数据上下文】 (喂给 AI 的原材料)
let dataContext = `【当前时间】${root.timestamp || '实时'}\n\n`;

// --- A. 持仓数据 ---
dataContext += `【1. 用户持仓数据】\n`;
if (portfolio.stocks && portfolio.stocks.length > 0) {
    portfolio.stocks.forEach(s => {
        dataContext += `- ${s.name}(${s.code}): 现价${s.price}, 涨跌${s.change_pct}%, 评级${s.rating_label}, 建议:${s.advice}\n`;
    });
} else {
    dataContext += "无持仓。\n";
}

// --- B. 市场大盘 ---
dataContext += `\n【2. 市场环境】\n`;
dataContext += `上证指数: ${sentiment.index_value}, 涨跌: ${sentiment.change_pct}%, 温度: ${sentiment.market_temperature}, 风险: ${sentiment.risk_level}\n`;

// --- C. 热点板块 (扩容：传入更多个股供 AI 筛选) ---
dataContext += `\n【3. 热点板块核心数据】(仅列出前5个强板块)\n`;
sectors.slice(0, 5).forEach((s, i) => {
    dataContext += `\n板块名称: ${s.name} (板块涨幅: ${s.change_pct}%)\n`;
    dataContext += `成分股数据:\n`;
    const stocks = s.leading_stocks || [];
    
    // 🔥 关键修改：传入前 10 只股票（量大管饱，防止 AI 没米下锅）
    stocks.slice(0, 10).forEach(st => {
        const vol = st.volume_ratio || 1.0;
        const rec = st.rating_label || st.recommendation || '⚪';
        dataContext += `  - 代码:${st.code}, 名称:${st.name}, 现价:${st.price}, 涨幅:${st.change_pct}%, 量比:${vol}, 评级:${rec}\n`;
    });
});

// --- D. 新闻数据 ---
dataContext += `\n【4. 近期重要新闻】\n`;
const newsItems = news.items || [];
newsItems.slice(0, 8).forEach(n => {
    dataContext += `- ${n.title}\n`;
});


// 4. 🧠 定义 AI 人设与模板 (System Prompt)
// 🔥 重点：在 Prompt 里强制要求输出 3-5 只
const systemPrompt = `你是一名拥有 20 年经验的 A 股顶级游资操盘手。
你的任务是基于用户提供的【原始数据】，撰写一份风格犀利的**《A股核心题材·深度狙击内参》**。

【报告格式严格要求】
请严格按照以下 Markdown 格式输出：

生成时间： [当前时间]
策略基准： 量价关系 + 题材共振
## 🚀 A股核心题材·深度狙击内参

### 持仓预警
*(遍历用户的持仓数据，对每一只股票给出操作建议)*
- **[股票名称] ([代码])**：现价 [价格] ([涨跌幅]%)，[简评，如"趋势完好，继续持有"]。

### 市场研判
- **市场温度**：[温度] (指数涨跌 [指数涨幅]%)
- **风险等级**：[风险等级]
- **操作建议**：[一句话建议]

### 机会挖掘
#### 板块作战表
*(选取数据中 **前 3 个** 最强的板块进行深度拆解)*

## 🚀 板块一：[板块名称] (涨幅 [板块涨幅]%)
**上涨逻辑**：[结合新闻或板块属性，推演一句话逻辑]

| 代码 | 名称 | 现价 | 角色 | 资金状态 | 狙击评级 |
| :--- | :--- | :--- | :--- | :--- | :--- |
| [代码] | [名称] | [价格] | [角色] | [资金] | [评级] |
*(⚠️ 核心指令：**每个板块必须列出 3-5 只个股！** 严禁只列 1 只。若数据中包含多只股票，请按涨幅排序填入表格。)*

**避坑指南**：[根据量比和涨幅给出一句风险提示]

*(重复上述结构生成板块二、板块三)*

### 新闻解读
*(挑选 3 条与上述板块最相关的新闻进行简评)*

### 交易师总结
- **仓位建议**：[给出建议]
- **重点观察**：[列出 1-2 只龙头]

---

【核心字段生成规则】
1. **角色判定**：涨幅第一且 >5% 为 "**🐲 龙头**"，其余为 "**⚔️ 跟风**"。
2. **资金状态**：量比 >1.5 "**放量**"，<0.8 "**缩量**"。
3. **个股数量**：**必须输出 3-5 行数据**，除非原始数据不足。`;

// 5. 输出
return {
    json: {
        messages: [
            { "role": "system", "content": systemPrompt },
            { "role": "user", "content": dataContext }
        ]
    }
};

// 完全重写 Binary 解析逻辑

let data = {};
let parseMethod = "未解析";
let debugInfo = {};

try {
    // ========== 方案1: 尝试从 Binary 读取 ==========
    if (items[0].binary) {
        const binaryKeys = Object.keys(items[0].binary);
        debugInfo.binaryKeys = binaryKeys;
        
        if (binaryKeys.length > 0) {
            const firstKey = binaryKeys[0];
            const binaryData = items[0].binary[firstKey];
            
            debugInfo.binaryMimeType = binaryData.mimeType;
            debugInfo.binaryHasData = !!binaryData.data;
            
            try {
                // N8N Binary 格式: { data: base64string, mimeType: string }
                const decoded = Buffer.from(binaryData.data, 'base64').toString('utf-8');
                debugInfo.decodedLength = decoded.length;
                debugInfo.decodedPreview = decoded.substring(0, 200);
                
                data = JSON.parse(decoded);
                parseMethod = "Binary (base64) - 成功";
            } catch (binaryError) {
                debugInfo.binaryError = binaryError.message;
                parseMethod = "Binary 解析失败: " + binaryError.message;
            }
        }
    }
    
    // ========== 方案2: 如果 Binary 失败，尝试 JSON 字段 ==========
    if (!data.market_sentiment && items[0].json) {
        const input = items[0].json;
        debugInfo.jsonKeys = Object.keys(input);
        
        // 尝试各种可能的字段
        const possibleFields = ['data', 'content', 'body', 'text', 'value'];
        
        for (const field of possibleFields) {
            if (input[field]) {
                try {
                    if (typeof input[field] === 'string') {
                        data = JSON.parse(input[field]);
                        parseMethod = `JSON - ${field}字段 (字符串)`;
                        break;
                    } else if (typeof input[field] === 'object' && input[field].market_sentiment) {
                        data = input[field];
                        parseMethod = `JSON - ${field}字段 (对象)`;
                        break;
                    }
                } catch (e) {
                    debugInfo[`${field}Error`] = e.message;
                }
            }
        }
        
        // 如果还没成功，检查 input 本身是否就是数据
        if (!data.market_sentiment && input.market_sentiment) {
            data = input;
            parseMethod = "JSON - 顶层对象";
        }
    }
    
} catch (error) {
    parseMethod = "致命错误: " + error.message;
    debugInfo.fatalError = error.stack;
}

// ========== 输出详细调试信息 ==========
console.log("========================================");
console.log("解析结果:", parseMethod);
console.log("是否成功:", !!data.market_sentiment);
console.log("调试信息:", JSON.stringify(debugInfo, null, 2));
console.log("========================================");

// ========== 如果还是失败，给出明确的错误提示 ==========
if (!data.market_sentiment) {
    return {
        json: {
            error: "数据解析失败",
            parseMethod: parseMethod,
            debugInfo: debugInfo,
            suggestion: "请检查: 1) Read File 节点的文件路径是否正确 2) 尝试在 Read File 节点的 Options 中添加 'As JSON' 选项 3) 或者使用 Execute Command 节点运行: cat /home/kk/n8n/market_data.json"
        }
    };
}

// ========== 以下是原有的数据处理逻辑 ==========
const sentiment = data.market_sentiment || {};
const mood = `市场温度: ${sentiment.market_temperature || "未知"} | 风险偏好: ${sentiment.risk_level || "平稳"} | 建议: ${sentiment.suggested_action || "暂无"}`;

let portfolioText = "";
if (data.my_portfolio && data.my_portfolio.stocks && data.my_portfolio.stocks.length > 0) {
    data.my_portfolio.stocks.forEach(stock => {
        const tech = stock.technical || {};
        const risks = tech.risks && tech.risks.length > 0 ? tech.risks.join(", ") : "无明显风险";
        const signals = tech.signals && tech.signals.length > 0 ? tech.signals.join(", ") : "无明显信号";
        const advice = stock.advice || "继续观察";
        
        portfolioText += `
- 股票名称: ${stock.name} (${stock.code})
  最新价格: ${stock.price} (涨幅: ${stock.change_pct}%)
  【量化建议】: ${advice}
  【RSI指标】: ${tech.rsi6 || "N/A"}
  【预警信号】: ${risks}
  【多头信号】: ${signals}
`;
    });
} else {
    portfolioText = "当前无持仓数据。";
}

let sectorText = "";
if (data.hot_sectors && data.hot_sectors.length > 0) {
    data.hot_sectors.slice(0, 5).forEach(sector => {
        sectorText += `\n> **${sector.name}** (板块涨幅: ${sector.change_pct}%, 热度: ${parseFloat(sector.heat_score || 0).toFixed(1)})\n`;
        if (sector.leading_stocks && sector.leading_stocks.length > 0) {
            sector.leading_stocks.slice(0, 3).forEach(stock => {
                const tech = stock.technical || {};
                const rec = stock.recommendation || "观察";
                sectorText += `  - ${stock.name}: ${stock.change_pct}% | 评级:${rec} | RSI:${tech.rsi6 || '-'}\n`;
            });
        }
    });
} else {
    sectorText = "无热点数据";
}

let newsText = "";
if (data.news_brief && data.news_brief.items) {
    data.news_brief.items.slice(0, 5).forEach((item, index) => {
        newsText += `${index + 1}. ${item.title}\n`;
    });
} else {
    newsText = "暂无新闻";
}

const prompt = `
你是一个专业的A股量化交易助手。请根据以下最新的市场实时数据，为用户生成一份《收盘复盘·持仓诊断报告》。

---数据开始---

【基本面概览】
${mood}

【我的持仓诊断】
${portfolioText}

【今日市场热点】
${sectorText}

【关键新闻】
${newsText}

---数据结束---

请按照 Markdown 格式输出报告：
## 🌞 收盘复盘·持仓诊断

### 1. 核心持仓策略
(根据【我的持仓诊断】给出操作建议)

### 2. 市场热点透视
(分析热点板块和资金流向)

### 3. 明日交易计划
(给出明天的操作建议)
`;

const dashscopeRequestBody = {
    model: "qwen-plus",
    input: {
        messages: [
            { role: "system", content: "你是专业的A股量化分析师，严格基于数据分析，不编造数据。" },
            { role: "user", content: prompt }
        ]
    },
    parameters: { result_format: "message", temperature: 0.7 }
};

return {
    json: {
        body: dashscopeRequestBody,
        prompt: prompt,
        debug: {
            parseMethod: parseMethod,
            parseSuccess: true,
            timestamp: data.timestamp || "未知",
            portfolioCount: data.my_portfolio?.stocks?.length || 0,
            portfolioStocks: data.my_portfolio?.stocks?.map(s => `${s.name}(${s.code}): ${s.change_pct}%`) || [],
            hotSectorsCount: data.hot_sectors?.length || 0,
            hotSectorsTop3: data.hot_sectors?.slice(0,3).map(s => s.name) || []
        }
    }
};
